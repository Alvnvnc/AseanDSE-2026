  \node[anchor=north west, text width=\slotw bp, align=left, text=ink,
        font=\smlfont\bfseries] at (\slotx,#3) {#6};
  \IfFileExists{gambar-siap/#5.png}{%
    \node[anchor=north west] at (\slotx,\sloty)
      {\includegraphics[width=\slotw bp, height=#4bp]{gambar-siap/#5}};
    \draw[rounded corners=3bp, draw=line, line width=0.6bp]
         (\slotx,\sloty) rectangle ++(\slotw,-#4);
  }{%
    \pgfmathsetmacro{\tw}{\slotw-32}%
    \draw[rounded corners=4bp, fill=soft, draw=line, dashed, line width=0.9bp]
         (\slotx,\sloty) rectangle ++(\slotw,-#4);
    \node[anchor=north west, fill=accent, text=white, rounded corners=2bp,
          inner xsep=5bp, inner ysep=3bp, font=\tnyfont\bfseries]
         at ({\slotx+10},{\sloty-10}) {SAC};

         at ({\slotx+10},{\sloty-34}) {\textbf{#7}\\[4bp]\textcolor{muted}{#8}};
    \node[anchor=south east, text=muted, font=\tnyfont]
         at ({\slotx+\slotw-10},{\sloty-#4+8}) {\texttt{gambar-sac/#5.png}};
  }%
}
